// New-Bit — Tablet do pátio (check-in por QR Code) // Depende de js/store.js e da biblioteca html5-qrcode (carregada via CDN no HTML). // O "código" lido do QR Code do cartão é o LOGIN do aluno (ex: ana.silva). document.addEventListener('DOMContentLoaded', () => { // O tablet precisa de um professor ou administrador logado para REGISTRAR presença. // A câmera e o botão ficam sempre visíveis; sem login, só mostramos um aviso. const sessaoCheckin = NewBitStore.getSession(); const logado = !!(sessaoCheckin && sessaoCheckin.token && ['admin', 'professor'].includes(sessaoCheckin.role)); const gate = document.getElementById('loginGate'); if (gate) gate.remove(); // aviso de login removido da tela const feedbackEl = document.getElementById('checkinFeedback'); const recentList = document.getElementById('recentList'); let ultimosHoje = []; let feedbackTimer = null; function mostrarFeedback(ok, mensagem) { feedbackEl.textContent = mensagem; feedbackEl.className = 'checkin-feedback ' + (ok ? 'is-ok' : 'is-err'); clearTimeout(feedbackTimer); feedbackTimer = setTimeout(() => { feedbackEl.className = 'checkin-feedback'; }, 4000); } function renderRecentList() { recentList.innerHTML = ultimosHoje.length === 0 ? '

Nenhuma presença registrada ainda hoje.

' : ultimosHoje.map(item => `
${NewBitStore.esc(item.nome)} ${item.hora}
`).join(''); } // O QR do cartão guarda o login do aluno, mas deixamos a leitura tolerante: // remove espaços/quebras de linha, ignora maiúsculas e aceita e-mail completo. function normalizarCodigo(texto) { return String(texto || '') .trim() .toLowerCase() .replace(/\s+/g, '') .split('@')[0]; } let processando = false; async function processarCodigo(codigoBruto) { const codigo = normalizarCodigo(codigoBruto); if (!codigo) { mostrarFeedback(false, 'QR Code vazio ou inválido.'); return; } if (!logado) { mostrarFeedback(false, 'Entre como professor/admin para registrar a presença.'); return; } if (processando) return; // evita registrar duas vezes enquanto o servidor responde processando = true; try { const resultado = await NewBitStore.registrarPresenca(codigo); mostrarFeedback(resultado.ok, resultado.message); if (resultado.ok && resultado.aluno) { const hora = new Date().toLocaleTimeString('pt-BR', { hour: '2-digit', minute: '2-digit' }); ultimosHoje.unshift({ nome: resultado.aluno.nome, hora }); renderRecentList(); } } catch (e) { mostrarFeedback(false, e.message || 'Não foi possível registrar a presença.'); } finally { processando = false; } } renderRecentList(); // ---------- Leitura por câmera (QR Code) ---------- const readerEl = document.getElementById('qr-reader'); let retryBtn = document.getElementById('cameraRetryBtn'); if (!retryBtn) { // Garante o botão mesmo que o checkin.html esteja desatualizado retryBtn = document.createElement('button'); retryBtn.id = 'cameraRetryBtn'; retryBtn.type = 'button'; retryBtn.className = 'btn-primary btn-block'; retryBtn.style.margin = '-6px 0 20px'; retryBtn.textContent = '📷 Abrir câmera e ler QR Code'; readerEl.insertAdjacentElement('afterend', retryBtn); } let scanner = null; let ultimoCodigoLido = null; let ultimoTimestamp = 0; function onScanSuccess(decodedText) { const agora = Date.now(); if (decodedText === ultimoCodigoLido && (agora - ultimoTimestamp) < 4000) return; ultimoCodigoLido = decodedText; ultimoTimestamp = agora; processarCodigo(decodedText); } function onScanFailure() { // Ignora frames sem QR Code (acontece várias vezes por segundo) } function erroCamera(mensagem) { readerEl.innerHTML = '

' + NewBitStore.esc(mensagem) + '

'; if (retryBtn) { retryBtn.textContent = '📷 Tentar de novo'; retryBtn.style.display = 'block'; } } // Caixa de leitura proporcional ao vídeo (um valor fixo quebra em telas pequenas) function calcularQrbox(largura, altura) { const lado = Math.floor(Math.min(largura, altura) * 0.7); return { width: Math.max(lado, 120), height: Math.max(lado, 120) }; } async function pararScanner() { if (!scanner) return; try { if (scanner.isScanning) await scanner.stop(); } catch (e) { /* já parado */ } try { scanner.clear(); } catch (e) { /* nada a limpar */ } scanner = null; } async function iniciarCamera() { if (retryBtn) retryBtn.style.display = 'none'; readerEl.innerHTML = '

Abrindo a câmera… se o navegador pedir, toque em "Permitir".

'; if (typeof Html5Qrcode === 'undefined') { erroCamera('Biblioteca de câmera não carregou (sem internet?). Use o campo manual abaixo.'); return; } // getUserMedia só funciona em HTTPS ou localhost (não em file:// nem http:// comum) if (!window.isSecureContext || !navigator.mediaDevices) { erroCamera('A câmera só funciona em página segura (https://). Use o campo manual abaixo.'); return; } await pararScanner(); readerEl.innerHTML = ''; scanner = new Html5Qrcode('qr-reader', { verbose: false }); const config = { fps: 10, qrbox: calcularQrbox, aspectRatio: 1.0 }; try { // 1ª tentativa: câmera traseira (a que o tablet usa para ler cartões) await scanner.start({ facingMode: 'environment' }, config, onScanSuccess, onScanFailure); } catch (e1) { console.warn('New-Bit: câmera traseira indisponível, tentando qualquer câmera.', e1); try { // 2ª tentativa: primeira câmera listada (notebooks, tablets só com frontal) const cameras = await Html5Qrcode.getCameras(); if (!cameras || !cameras.length) throw new Error('Nenhuma câmera encontrada.'); await scanner.start(cameras[0].id, config, onScanSuccess, onScanFailure); } catch (e2) { console.error('New-Bit: não foi possível iniciar a câmera.', e2); const negada = /permission|denied|notallowed/i.test(String(e2 && (e2.name || e2.message || e2))); erroCamera(negada ? 'Permissão da câmera negada. Libere a câmera nas configurações do navegador e toque em "Tentar de novo".' : 'Não foi possível acessar a câmera. Use o campo manual abaixo ou toque em "Tentar de novo".'); } } } if (retryBtn) retryBtn.addEventListener('click', iniciarCamera); // Libera a câmera ao sair da página / trocar de aba no tablet window.addEventListener('pagehide', pararScanner); // ---------- Entrada manual (sem câmera) ---------- const manualBtn = document.getElementById('manualBtn'); const manualInput = document.getElementById('manualLogin'); if (manualBtn && manualInput) { manualBtn.addEventListener('click', () => { const codigo = manualInput.value.trim(); if (!codigo) { mostrarFeedback(false, 'Digite o login do aluno.'); return; } processarCodigo(codigo); manualInput.value = ''; }); manualInput.addEventListener('keydown', (e) => { if (e.key === 'Enter') manualBtn.click(); }); } });
